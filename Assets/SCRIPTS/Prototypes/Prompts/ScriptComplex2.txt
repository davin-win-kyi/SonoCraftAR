[Example]
- **Input:** 'A Sound bar that increases/decreases the amount filled in based on sound volume.'  
- **Expected Behavior:** The generated script Draws a sound bar that is filled based on the volume values received in `soundData`. It will dynamically 
    change its volume as new values come in. 
- **Example Script:** 
using UnityEngine;
using Shapes;

public class DynamicVolumeBar : ImmediateModeShapeDrawer
{
    // explains the key considerations for implementing this class
    public string summary = “The volume bar should feature a visually appealing design that incorporates smooth animations or gradients to enhance its appearance. It should use clear visual cues, such as a dynamic fill to reflect the current volume level. I can use two rectangles.”;	

    [Range(0, 1)]
    public float fillAmount = 1;
    // a gradient is visually aesthetically pleasing
    public Gradient colorGradient;
    public string title = "Volume Bar";

    // this controls how quickly the animation interpolates
    private float smoothSpeed = 5f;

    private float targetFillAmount = 1f;

    // this function is constantly called outside to receive real sound data input
    public void <GPT_FUNCTION_NAME>(object[] soundData)
    {
        if (soundData.Length >= 3)
        {
            string classification = soundData[0] as string;
            float pitch = (float)soundData[1];
            float distance = (float)soundData[2];
	
            float rawVolume = Mathf.Clamp01(1f / (distance + 1f)); //acc to gpt, volume is 1/(distance + 1)

            targetFillAmount = rawVolume;
        }
    }

    // this constantly updates fillAmount with Math.Lerp, creating smooth animations for the sound bar
    private void FixedUpdate()
    {
        // interpolating`fillAmount` towards `targetFillAmount` is the key to smoother animations
        fillAmount = Mathf.Lerp(fillAmount, targetFillAmount, Time.deltaTime * smoothSpeed);
    }

    // this function constantly updates the filled amount of the volume bar
    public override void DrawShapes(Camera cam)
    {
        using (Draw.Command(cam))
        {
            // making sure the gradient is initialized
            if (colorGradient == null)
            {
                colorGradient = CreateGradient();
            }


            Rect rect = new Rect(0, 0, 8, 1);

            // black background
            Draw.Rectangle(rect, 0.3f, Color.black);

            // the colored bar
            Rect fillRect = Inset(rect, 1);
            fillRect.width *= fillAmount;
            Draw.Rectangle(fillRect, colorGradient.Evaluate(fillAmount));

            // Draw the title
            Draw.FontSize = 10;
            Vector2 topLeft = new Vector2(rect.xMin, rect.yMax + 1f);
            Draw.Text(topLeft, title, TextAlign.BaselineLeft);
        }
    }

    // this optional function creates and returns a Rect object for the sound bar’s outline
    Rect Inset(Rect r, float amount)
    {
        return new Rect(r.x + 0.25f, r.y + 0.25f, r.width - amount * 0.5f, r.height - amount * 0.5f);
    }

    // this optional function creates and returns a Gradient for the sound bar 
    Gradient CreateGradient()
    {
        Gradient gradient = new Gradient();

        // Set the color keys
        GradientColorKey[] colorKeys = new GradientColorKey[2];
        colorKeys[0].color = Color.yellow; // Start color
        colorKeys[0].time = 0.0f; // At the start of the gradient
        colorKeys[1].color = Color.cyan; // End color
        colorKeys[1].time = 1.0f; // At the end of the gradient

        // Set the alpha keys
        GradientAlphaKey[] alphaKeys = new GradientAlphaKey[2];
        alphaKeys[0].alpha = 1.0f; // Fully opaque at the start
        alphaKeys[0].time = 0.0f;
        alphaKeys[1].alpha = 1.0f; // Fully opaque at the end
        alphaKeys[1].time = 1.0f;

        // Assign the keys to the gradient
        gradient.SetKeys(colorKeys, alphaKeys);

        return gradient;
    }
}
