using UnityEngine;
using UnityEngine.UI;
using System;

/// <summary>
/// Visualizes a green circle on a canvas, adjusting its position based on input sound data.
/// </summary>
public class GreenCircleVisualizer : MonoBehaviour
{
    public GameObject visualizerObject;
    public RectTransform visualizerTransform;
    public Canvas canvas;
    public Color visualizerColor = Color.green;

    private void Start()
    {
        SetupCanvas();
        SetupVisualizerObject();
    }

    /// <summary>
    /// Sets up a Canvas if none is assigned.
    /// </summary>
    private void SetupCanvas()
    {
        if (canvas == null)
        {
            GameObject canvasGO = new GameObject("VisualizerCanvas");
            canvas = canvasGO.AddComponent<Canvas>();
            canvas.renderMode = RenderMode.ScreenSpaceOverlay;
            canvas.sortingOrder = 100;

            CanvasScaler canvasScaler = canvasGO.AddComponent<CanvasScaler>();
            canvasScaler.uiScaleMode = CanvasScaler.ScaleMode.ScaleWithScreenSize;
            canvasScaler.referenceResolution = new Vector2(1920, 1080);
        }
    }

    /// <summary>
    /// Sets up the visualizer object and its RectTransform.
    /// </summary>
    private void SetupVisualizerObject()
    {
        if (visualizerObject == null)
        {
            visualizerObject = new GameObject("VisualizerCircle");
            visualizerObject.transform.SetParent(canvas.transform, false);

            Image image = visualizerObject.AddComponent<Image>();
            image.color = visualizerColor;
            image.sprite = CreateCircleSprite();
        }

        if (visualizerTransform == null)
        {
            visualizerTransform = visualizerObject.GetComponent<RectTransform>();
            if (visualizerTransform == null)
            {
                Debug.LogError("Visualizer transform is missing; attempting to create it.");
                visualizerTransform = visualizerObject.AddComponent<RectTransform>();
            }
        }

        // Set initial size and position
        float visualizerSize = Screen.width * 0.05f; // 5% of screen width
        visualizerTransform.sizeDelta = new Vector2(visualizerSize, visualizerSize);
        visualizerTransform.anchoredPosition = Vector2.zero;
    }

    /// <summary>
    /// Creates a circular sprite using a dynamically generated texture.
    /// </summary>
    private Sprite CreateCircleSprite()
    {
        int diameter = 128;
        Texture2D texture = new Texture2D(diameter, diameter);
        Color[] colors = new Color[diameter * diameter];

        float radius = diameter / 2f;
        Vector2 center = new Vector2(radius, radius);

        for (int y = 0; y < diameter; y++)
        {
            for (int x = 0; x < diameter; x++)
            {
                Vector2 pos = new Vector2(x, y);
                float dist = Vector2.Distance(center, pos);
                colors[y * diameter + x] = dist <= radius ? visualizerColor : Color.clear;
            }
        }

        texture.SetPixels(colors);
        texture.Apply();

        return Sprite.Create(texture, new Rect(0, 0, diameter, diameter), new Vector2(0.5f, 0.5f));
    }

    /// <summary>
    /// Updates the visualizer's position based on sound data.
    /// Assumes soundData contains: [classification, pitch, distance].
    /// </summary>
    public void GeneratedFunction(object[] soundData)
    {
        if (soundData == null || soundData.Length < 3) return;
        if (!(soundData[2] is float distance)) return;

        // Adjust y-axis position based on distance (illustrative use)
        float newYPosition = Mathf.Lerp(-Screen.height / 2, Screen.height / 2, distance);
        visualizerTransform.anchoredPosition = new Vector2(visualizerTransform.anchoredPosition.x, newYPosition);
    }
}
